cmake_minimum_required(VERSION 3.20)
project(dascript_flatbuffers_tools LANGUAGES CXX)
set(FLATBUFFERS_SOURCE_DIR "" CACHE PATH "Official FlatBuffers source checkout")
set(DASLANG_EXECUTABLE "daslang" CACHE STRING "daScript interpreter")
if(NOT EXISTS "${FLATBUFFERS_SOURCE_DIR}/src/flatc_main.cpp")
  message(FATAL_ERROR "Set FLATBUFFERS_SOURCE_DIR to a FlatBuffers source checkout")
endif()
set(FLATBUFFERS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(FLATBUFFERS_INSTALL OFF CACHE BOOL "" FORCE)
set(FLATBUFFERS_BUILD_FLATC ON CACHE BOOL "" FORCE)
add_subdirectory("${FLATBUFFERS_SOURCE_DIR}" flatbuffers)
# Generate a replacement entry point in the build tree; the dependency is untouched.
file(READ "${FLATBUFFERS_SOURCE_DIR}/src/flatc_main.cpp" MAIN)
string(REPLACE "int main(int argc, const char* argv[]) {"
  "namespace flatbuffers { std::unique_ptr<CodeGenerator> NewDascriptCodeGenerator(); }\nint main(int argc, const char* argv[]) {" MAIN "${MAIN}")
string(REPLACE "flatbuffers::FlatCompiler flatc(params);"
  "flatbuffers::FlatCompiler flatc(params);\nflatc.RegisterCodeGenerator(flatbuffers::FlatCOption{\"\", \"dascript\", \"\", \"Generate daScript through the shared BFBS backend\"}, flatbuffers::NewDascriptCodeGenerator());" MAIN "${MAIN}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/dascript_flatc_main.cpp" "${MAIN}")
get_target_property(FLATC_SOURCES flatc SOURCES)
list(FILTER FLATC_SOURCES EXCLUDE REGEX "flatc_main.cpp$")
set_property(TARGET flatc PROPERTY SOURCES "${FLATC_SOURCES}")
target_sources(flatc PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/dascript_flatc_main.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/dascript_generator.cpp")
target_include_directories(flatc PRIVATE "${FLATBUFFERS_SOURCE_DIR}/src")
target_compile_features(flatc PRIVATE cxx_std_17)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(DAS_FLATBUFFERS_ROOT "${REPO}" CACHE PATH "Runtime location used by the flatc adapter")
target_compile_definitions(flatc PRIVATE DASLANG_DEFAULT="${DASLANG_EXECUTABLE}" DAS_GENERATOR_DEFAULT="${DAS_FLATBUFFERS_ROOT}/tools/generate.das" DAS_ANNOTATIONS_DEFAULT="${DAS_FLATBUFFERS_ROOT}/src/annotations.das")
add_executable(fb_verify_cpp verify.cpp)
target_link_libraries(fb_verify_cpp PRIVATE flatbuffers)
target_compile_features(fb_verify_cpp PRIVATE cxx_std_17)
